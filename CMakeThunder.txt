# =============================================================================
# CUDA Autoencoder for CIFAR-10 Feature Learning with ThunderSVM Integration
# =============================================================================
# NOTE: This is an alternative CMake configuration for using ThunderSVM
# To use this file, replace CMakeLists.txt with this file:
#   cp CMakeThunder.txt CMakeLists.txt
# =============================================================================
cmake_minimum_required(VERSION 3.18)
project(CUDAAutoencoder CXX C)

# =============================================================================
# C++ Standard Configuration
# =============================================================================
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

# =============================================================================
# Build Options
# =============================================================================
option(ENABLE_CUDA "Enable CUDA support (requires NVIDIA GPU)" ON)

# =============================================================================
# Find Required Packages
# =============================================================================
# find_package(OpenMP REQUIRED)

# CUDA Support
if(ENABLE_CUDA)
    find_package(CUDAToolkit)
    if(CUDAToolkit_FOUND)
        message(STATUS "CUDA Toolkit found - GPU support enabled")
        enable_language(CUDA)
        set(CUDA_AVAILABLE TRUE)
    else()
        message(WARNING "CUDA requested but not found - building CPU-only version")
        set(CUDA_AVAILABLE FALSE)
    endif()
else()
    message(STATUS "Building CPU-only version (set -DENABLE_CUDA=ON for GPU support)")
    set(CUDA_AVAILABLE FALSE)
endif()

# Project Directories
include_directories(${CMAKE_SOURCE_DIR}/include)
include_directories(${CMAKE_SOURCE_DIR}/src)
include_directories(${CMAKE_SOURCE_DIR}/external/thundersvm/include)
link_directories(${CMAKE_SOURCE_DIR}/external/thundersvm/build/lib)

# Define project root path for use in code
add_compile_definitions(PROJECT_ROOT_DIR="${CMAKE_SOURCE_DIR}")
######################################### set source files #########################################
set(UTILS_SOURCES
    src/utils/logger.cpp
    src/utils/image_utils.cpp
    src/utils/memory_tracker.cpp
)

set(DATA_SOURCES
    src/data/cifar10_loader.cpp
)

set(SVM_SOURCES
    external/svm.cu
    external/libsvm/svm.cpp
)

set(THUNDER_SVM_SOURCES
    external/thunder_svm.cu
)

set(CPU_LAYER_SOURCES
    src/layers/conv2d_cpu.cpp
    src/layers/relu_cpu.cpp
    src/layers/maxpool_cpu.cpp
    src/layers/upsample_cpu.cpp
)

set(CPU_MODEL_SOURCES
    src/models/autoencoder_cpu.cpp
)

######################################### Set up executables #########################################
# =============================================================================
# CPU Training
# =============================================================================
add_executable(train_cpu
    src/train_cpu.cpp
    ${DATA_SOURCES}
    ${CPU_LAYER_SOURCES}
    ${CPU_MODEL_SOURCES}
    ${UTILS_SOURCES}
)

# =============================================================================
# CPU Feature Extraction
# =============================================================================
add_executable(extract_cpu
    src/models_svm/extract_cpu.cpp
    ${DATA_SOURCES}
    ${CPU_LAYER_SOURCES}
    ${CPU_MODEL_SOURCES}
)

# =============================================================================
# SVM Training (CPU)
# =============================================================================
add_executable(train_svm_cpu
    src/models_svm/train_svm_cpu.cpp
    ${DATA_SOURCES}
    external/svm.cpp
)

# Backward-compatible alias (older README/scripts)
add_executable(extract_features_cpu
    src/models_svm/extract_cpu.cpp
    ${DATA_SOURCES}
    ${CPU_LAYER_SOURCES}
    ${CPU_MODEL_SOURCES}
)
